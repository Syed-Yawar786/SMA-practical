NO. 07 Aim: Apply Particle Swarm Optimization to determine optimal
sensor placement in a network.

import numpy as np
import matplotlib.pyplot as plt

grid_size = 20

xs = np.arange(grid_size)
ys = np.arange(grid_size)

grid_points = np.array([[x, y] for x in xs for y in ys])

K = 5

sensor_range = 5.0

def fitness(position):
    sensors = position.reshape(K, 2)

    distances = np.linalg.norm(
        grid_points[:, np.newaxis, :] - sensors[np.newaxis, :, :],
        axis=2
    )

    covered = np.any(distances <= sensor_range, axis=1)

    return 1.0 - np.mean(covered)


n_particles = 30
n_iterations = 100

w = 0.7
c1 = 1.5
c2 = 1.5

dim = 2 * K

positions = np.random.uniform(
    0, grid_size - 1, (n_particles, dim)
)

velocities = np.random.uniform(-1, 1, (n_particles, dim))

pbest_positions = positions.copy()

pbest_scores = np.array([
    fitness(p) for p in positions
])

best_index = np.argmin(pbest_scores)

gbest_position = pbest_positions[best_index].copy()

gbest_score = pbest_scores[best_index]


for iteration in range(n_iterations):

    for i in range(n_particles):

        r1 = np.random.rand(dim)
        r2 = np.random.rand(dim)

        velocities[i] = (
            w * velocities[i]
            + c1 * r1 * (pbest_positions[i] - positions[i])
            + c2 * r2 * (gbest_position - positions[i])
        )

        positions[i] = positions[i] + velocities[i]

        positions[i] = np.clip(
            positions[i], 0, grid_size - 1
        )

        score = fitness(positions[i])

        if score < pbest_scores[i]:

            pbest_scores[i] = score

            pbest_positions[i] = positions[i].copy()

        if score < gbest_score:

            gbest_score = score

            gbest_position = positions[i].copy()

    if iteration % 10 == 0:

        print(
            f"Iteration {iteration:3d} | "
            f"Best uncovered fraction: {gbest_score:.4f}"
        )


best_sensors = gbest_position.reshape(K, 2)

print("\nSENSOR POSITIONS (x, y)")

for i, (x, y) in enumerate(best_sensors, start=1):
    print(f"Sensor {i}: ({x:.2f}, {y:.2f})")

coverage = (1 - gbest_score) * 100

print(f"\nCoverage achieved: {coverage:.2f}%")

print(f"Uncovered fraction: {gbest_score:.4f}")


fig, ax = plt.subplots(figsize=(7, 7))

ax.scatter(
    grid_points[:, 0],
    grid_points[:, 1],
    s=8,
    label="Grid points"
)

ax.scatter(
    best_sensors[:, 0],
    best_sensors[:, 1],
    marker="*",
    s=180,
    label="Sensors"
)

for x, y in best_sensors:

    circle = plt.Circle(
        (x, y),
        sensor_range,
        fill=False,
        linewidth=1.5
    )

    ax.add_patch(circle)

ax.set_xlim(-1, grid_size)
ax.set_ylim(-1, grid_size)

ax.set_aspect("equal")

ax.set_title(
    f"Optimal Sensor Placement (Coverage: {coverage:.2f}%)"
)

ax.set_xlabel("X coordinate")
ax.set_ylabel("Y coordinate")

ax.legend()
ax.grid(True)

plt.show()
